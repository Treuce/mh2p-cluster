cmake_minimum_required(VERSION 3.16)
project(mib2p_lifecycle_tests C)
enable_testing()
add_executable(lifecycle_native_tests lifecycle_native_tests.c)
target_include_directories(lifecycle_native_tests PRIVATE ../src)
if(MSVC)
  target_compile_options(lifecycle_native_tests PRIVATE /W4)
else()
  target_compile_options(lifecycle_native_tests PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME lifecycle_native COMMAND lifecycle_native_tests)
add_executable(receiver_ring_tests receiver_ring_tests.c)
target_include_directories(receiver_ring_tests PRIVATE ../src)
add_test(NAME receiver_ring COMMAND receiver_ring_tests)
add_executable(lifecycle_native_peer lifecycle_native_peer.c)
target_include_directories(lifecycle_native_peer PRIVATE ../src)
if(MSVC)
  target_compile_definitions(lifecycle_native_tests PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_definitions(lifecycle_native_peer PRIVATE _CRT_SECURE_NO_WARNINGS)
  add_library(lifecycle_daemon_syntax OBJECT lifecycle_daemon_compile.c)
  target_include_directories(lifecycle_daemon_syntax PRIVATE ../src)
  target_compile_definitions(lifecycle_daemon_syntax PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(lifecycle_daemon_syntax PRIVATE /W4 /WX)
endif()
